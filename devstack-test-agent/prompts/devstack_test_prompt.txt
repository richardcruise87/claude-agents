

## Step 8: Cleanup Test Resources
**CRITICAL:** Clean up ALL test resources created during testing.

```bash
source {openrc_file}

# Delete test load balancers (using unique prefix)
for lb_id in $(openstack loadbalancer list --name {resource_prefix} -f value -c id); do
  echo "Deleting load balancer: $lb_id"
  openstack loadbalancer delete --cascade $lb_id

  # Wait for deletion
  while openstack loadbalancer show $lb_id 2>/dev/null; do
    sleep 2
  done
done

# Delete any test servers
for server_id in $(openstack server list --name {resource_prefix} -f value -c ID); do
  echo "Deleting server: $server_id"
  openstack server delete $server_id
done

# Verify cleanup
echo "Remaining test resources:"
openstack loadbalancer list --name {resource_prefix}
openstack server list --name {resource_prefix}
```
